# --------------------------------------------
# 1. IMPORTAR LIBRERÍAS
# --------------------------------------------
import numpy as np
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

# --------------------------------------------
# 2. CARGAR DATASET
# --------------------------------------------
# load_iris() trae un dataset de 150x4 registros flores listo para usar
# 150 registro con 4 características

iris = load_iris()

# X = características (inputs)
# y = etiquetas (clases)
X = iris.data
y = iris.target

print("=== INFORMACIÓN DEL DATASET ===")
print("Shape de X:", X.shape)  # (150, 4)
print("Shape de y:", y.shape)  # (150,)

# --------------------------------------------
# 3. EXPLORAR DATOS (opcional pero recomendado)
# --------------------------------------------
print("\nPrimeros 5 datos:")
print(X[:5])

print("\nPrimeras 5 etiquetas:")
print(y[:5])

# --------------------------------------------
# 4. DIVISIÓN DE DATOS (TRAIN / TEST)
# --------------------------------------------
# 80% entrenamiento, 20% prueba
X_train, X_test, y_train, y_test = train_test_split(
    X, y,
    test_size=0.2,      # 20% para test
    random_state=42     # para reproducibilidad
)

print("\n=== DIVISIÓN DE DATOS ===")
print("X_train:", X_train.shape)
print("X_test:", X_test.shape)
print("y_train:", y_train.shape)
print("y_test:", y_test.shape)

# --------------------------------------------
# 5. NORMALIZACIÓN (ESCALADO)
# --------------------------------------------
# IMPORTANTE:
# Se ajusta SOLO con datos de entrenamiento
scaler = StandardScaler()

# Aprende media y desviación estándar
X_train = scaler.fit_transform(X_train)

# Aplica lo aprendido al test (SIN aprender de test)
X_test = scaler.transform(X_test)

# --------------------------------------------
# 6. VERIFICACIÓN FINAL
# --------------------------------------------
print("\n=== DATOS NORMALIZADOS ===")
print("Ejemplo X_train:")
print(X_train[:5])

print("\nMedia aproximada (debe ser cercana a 0):")
print(np.mean(X_train, axis=0))

print("\nDesviación estándar (debe ser cercana a 1):")
print(np.std(X_train, axis=0))

# --------------------------------------------
# 3. CONVERTIR ETIQUETAS A ONE-HOT
# --------------------------------------------
# Ejemplo: 1 → [0,1,0]
y_train_cat = to_categorical(y_train)
y_test_cat = to_categorical(y_test)

print("Ejemplo etiqueta original:", y_train[0])
print("Ejemplo one-hot:", y_train_cat[0])

# --------------------------------------------
# 4. CREAR MODELO MLP MÍNIMO
# --------------------------------------------
model = Sequential()

# Capa oculta (16 neuronas)
model.add(Dense(16, activation='relu', input_shape=(4,)))

# Capa de salida (3 clases)
model.add(Dense(3, activation='softmax'))

# Mostrar estructura
print("\n=== ESTRUCTURA DEL MODELO ===")
model.summary()

# --------------------------------------------
# 5. COMPILAR MODELO
# --------------------------------------------
model.compile(
    optimizer='adam',                 # optimizador
    loss='categorical_crossentropy',  # función de pérdida
    metrics=['accuracy']              # métrica
)

# --------------------------------------------
# 6. ENTRENAR MODELO
# --------------------------------------------
print("\n=== ENTRENANDO MODELO ===")

history = model.fit(
    X_train, y_train_cat,
    epochs=50,            # número de iteraciones
    batch_size=8,         # tamaño de lote
    validation_split=0.2, # 20% validación
    verbose=1
)

# --------------------------------------------
# 7. EVALUACIÓN FINAL
# --------------------------------------------
print("\n=== EVALUACIÓN EN TEST ===")

loss, accuracy = model.evaluate(X_test, y_test_cat)

print("Loss:", loss)
print("Accuracy:", accuracy)

# --------------------------------------------
# 8. PREDICCIONES
# --------------------------------------------
print("\n=== PREDICCIONES ===")

predicciones = model.predict(X_test)

# Mostrar ejemplo
print("\nProbabilidades (primer dato):")
print(predicciones[0])

print("\nClase predicha:")
print(np.argmax(predicciones[0]))

print("Clase real:")
print(y_test[0])
