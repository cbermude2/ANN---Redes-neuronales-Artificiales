# --------------------------------------------
# 1. IMPORTAR LIBRERÍAS
# --------------------------------------------
import numpy as np
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

# --------------------------------------------
# 2. CARGAR DATASET
# --------------------------------------------
# load_iris() trae un dataset de flores listo para usar
iris = load_iris()

# X = características (inputs)
# y = etiquetas (clases)
X = iris.data
y = iris.target

print("=== INFORMACIÓN DEL DATASET ===")
print("Shape de X:", X.shape)  # (150, 4)
print("Shape de y:", y.shape)  # (150,)

# --------------------------------------------
# 3. EXPLORAR DATOS (opcional pero recomendado)
# --------------------------------------------
print("\nPrimeros 5 datos:")
print(X[:5])

print("\nPrimeras 5 etiquetas:")
print(y[:5])

# --------------------------------------------
# 4. DIVISIÓN DE DATOS (TRAIN / TEST)
# --------------------------------------------
# 80% entrenamiento, 20% prueba
X_train, X_test, y_train, y_test = train_test_split(
    X, y,
    test_size=0.2,      # 20% para test
    random_state=42     # para reproducibilidad
)

print("\n=== DIVISIÓN DE DATOS ===")
print("X_train:", X_train.shape)
print("X_test:", X_test.shape)
print("y_train:", y_train.shape)
print("y_test:", y_test.shape)

# --------------------------------------------
# 5. NORMALIZACIÓN (ESCALADO)
# --------------------------------------------
# IMPORTANTE:
# Se ajusta SOLO con datos de entrenamiento
scaler = StandardScaler()

# Aprende media y desviación estándar
X_train = scaler.fit_transform(X_train)

# Aplica lo aprendido al test (SIN aprender de test)
X_test = scaler.transform(X_test)

# --------------------------------------------
# 6. VERIFICACIÓN FINAL
# --------------------------------------------
print("\n=== DATOS NORMALIZADOS ===")
print("Ejemplo X_train:")
print(X_train[:5])

print("\nMedia aproximada (debe ser cercana a 0):")
print(np.mean(X_train, axis=0))

print("\nDesviación estándar (debe ser cercana a 1):")
print(np.std(X_train, axis=0))
